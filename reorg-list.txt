  try {
    await document.fonts?.ready
    /* 字体字节不进草稿，进页面时可能没取成功 —— 导出前再确认一次，

    }

    const { blob, lost, font } = await cover.svgToPng(svg, fontCss)
    if (!blob) throw new Error('画布导出失败')
    cover.download(blob, cover.pngName(s))
    if (lost) toast.error(`有 ${lost} 张图没能烘进导出图（网络 / 跨域拿不到），导出图里会缺这几张`)
    else if (font) toast.success(`导出成功（已嵌入字体：${s.fontName}）`)
    else toast.success('导出成功')
  } catch (err) {
    toast.error(`导出失败：${err.message}`)
  } finally {
    exporting.value = false
  }

/* ------------------------------------------------------------------ 字体重取 */

/* 草稿只留「选了哪个字体」（几兆的 base64 不进草稿）：
   优先按 fontFileName 找；老草稿只记了名字就按名字找 —— 名字与文件名都以 fonts.json 为准 */
async function ensureFont() {
  if (s.fontName && s.fontData) return true

  try {
    if (!fontList.value.length) fontList.value = await fonts.fetchFontList()
  } catch {
    /* 清单取不到就先按手头的信息试，实在没有就由调用方提示 */
  }

  const item = fontList.value.find((f) => f.file === s.fontFileName)
    || fontList.value.find((f) => f.name === s.fontName)
  if (!item) return false

  fontStatus.value = 'loading'
  try {
    const data = await fonts.fetchFontData(item.file)
    if (!(await fonts.installFont({ name: item.name, data }))) throw new Error('字体解析失败')

    s.fontFileName = item.file
    s.fontName = item.name
    s.fontData = data
    loaded.value = item.file
    fontFile.value = item.file
    fontStatus.value = ''
    return true
  } catch (err) {
    fonts.dropFont()
    s.fontData = ''
    s.fontName = ''
    fontStatus.value = 'error'
    ElMessage.error(`字体「${item.name}」没能加载：${err.message}，导出会用默认字体`)
    return false
  }
}

async function exportPng() {
  const svg = svgEl.value
  if (!svg || exporting.value) return
  exporting.value = true

  try {
    await document.fonts?.ready
    /* 字体字节不进草稿，进页面时可能没取成功 —— 导出前再确认一次，
       拿不到就明说，别静默给一张默认字体的图 */
    if ((s.fontName || s.fontFileName) && !s.fontData) await ensureFont()

    /* 字体跟着 SVG 一起进导出图：CSS 在 bakeSvg 里用 createElementNS 内嵌 */
    const fontCss = s.fontName && s.fontData ? fonts.fontFaceCss(s.fontName, s.fontData) : ''
    if (!fontCss && (s.fontName || s.fontFileName)) {
      ElMessage.error('字体没能加载，这次导出用的是默认字体')
    }

    const { blob, lost, font } = await cover.svgToPng(svg, fontCss)
    if (!blob) throw new Error('画布导出失败')
    cover.download(blob, cover.pngName(s))
    if (lost) ElMessage.error(`有 ${lost} 张图没能烘进导出图（网络 / 跨域拿不到），导出图里会缺这几张`)
    else if (font) ElMessage.success(`导出成功（已嵌入字体：${s.fontName}）`)
    else ElMessage.success('导出成功')
  } catch (err) {
    ElMessage.error(`导出失败：${err.message}`)
  } finally {
    exporting.value = false
  }
}
