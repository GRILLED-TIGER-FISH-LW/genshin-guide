      /* reactive 代理不能直接进 IDB：嵌套的数组 / 对象还是 Proxy，structured
         clone 抛 DataCloneError，被下面那句 catch 吞掉就是「草稿永远不更新」——

    }
  }, 500)

async function bakeSvg(svgEl) {
  const clone = svgEl.cloneNode(true)

  /* 参考线（虚线）是预览里对齐用的，导出图一律不要 —— 跟预览开没开无关 */
  for (const el of clone.querySelectorAll('[data-guides]')) el.remove()

  /* 自包含：外链图一张都不能留 —— SVG 当图片渲染时不加载任何外部资源 */
  let lost = 0
  for (const img of clone.querySelectorAll('image')) {
    const href = img.getAttribute('href')
    if (!href || /^data:/i.test(href)) continue
    const baked = await urlToDataURL(href)
    if (/^data:/i.test(baked)) img.setAttribute('href', baked)
    else {
      /* 网络 / 跨域拿不到：留着也是空白，删掉并记账，由调用方提示用户 */
      img.remove()
      lost += 1
    }
  }

  /* 预览靠 CSS 撑满，节点本身没有宽高；导出必须给上，
     否则 Image 会按 300×150 的默认尺寸栅格化，出来一张糊图 */
  clone.setAttribute('xmlns', 'http://www.w3.org/2000/svg')
  clone.setAttribute('width', String(W))
  clone.setAttribute('height', String(H))

  return { svg: new XMLSerializer().serializeToString(clone), lost }
}

/* ------------------------------------------------------------------ 字体重取 */

/* 草稿只记得「选了哪个字体」（几兆的 base64 不进草稿），
   进页面与导出前都按文件名去 public/fonts/ 取回来再装 —— 名字以清单为准，不信草稿 */
async function ensureFont() {
  if (s.fontName && s.fontData) return true
  const file = s.fontFileName
  if (!file) return false

  fontStatus.value = 'loading'
  try {
    if (!fontList.value.length) fontList.value = await fonts.fetchFontList()
    const item = fontList.value.find((f) => f.file === file)
    if (!item) throw new Error('字体清单里没有这个文件')

    const data = await fonts.fetchFontData(file)
    if (!(await fonts.installFont({ name: item.name, data }))) throw new Error('字体解析失败')

    s.fontName = item.name
    s.fontData = data
    loaded.value = file
    fontFile.value = file
    fontStatus.value = ''
    return true
  } catch (err) {
    fonts.dropFont()
    s.fontData = ''
    s.fontName = ''
    fontStatus.value = 'error'
    toast.error(`字体「${file}」没能加载：${err.message}，导出会用默认字体`)
    return false
  }
}

## 2026 年 11 月

### 2026-11-10 · 视频封面四修：草稿存不进、字体不重取、Logo 旧路径、参考线进了导出

- **起因**：用户报「上次选过的字体与 Logo，再进来导出图却是默认字体、Logo 也没了」，并指出「因为没有重加载」。顺着查下去，根因比表面更靠前：视频封面的草稿**从来没存进去过**。
- **经过**：`save()` 把 `reactive` 状态直接 `put()` 进 IndexedDB，嵌套的数组 / 对象仍是 Proxy，structured clone 抛 `DataCloneError`，被那句「存不下就算了」吞掉 —— 每次进页面读到的都是**上一次成功写入的旧草稿**（工具页时代那份：Logo 还是 `shared/assets/video-cover/…` 的老路径，字体的 base64 也没带），预览与导出的差异全从这儿来。四处一起改：① 草稿先过一次 JSON 变成纯数据再存，字体的几兆 base64 干脆不进草稿；② 进页面与导出前都按文件名去 `public/fonts/` 重取字体字节并装好（`ensureFont()`），取不到就明说「这次导出用默认字体」，不再静默；③ Logo 按文件名对回当前目录（`healLogo()`），老路径自愈；④ 参考线（虚线）只是预览对齐用的，导出前从克隆出来的 SVG 里删掉（模板上打 `data-guides` 标记）。
- **结果**：草稿真的会存了、字体与 Logo 每次进来都会重新取好；导出图里不再出现参考线；另外把「烘不进去的图」变成显式提示 —— 有图因跨域 / 网络没能换成 data URL 时，会告诉用户缺几张，而不是默默给一张缺图的成品。
