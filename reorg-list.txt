/* 数字框在编辑期间不回灌 —— 否则输 9999 会被立刻改成 3840，光标也跳 */
const editing = ref(false)

const clamp = (n) => Math.min(props.max, Math.max(props.min, n))

function onNumber(e) {
  const raw = e.target.value
  if (raw === '' || raw === '-') return
  const n = Number(raw)
  if (!Number.isFinite(n)) return
  model.value = clamp(n)
}

/* 失焦才把框里归位到真正生效的值 */
function endEdit(e) {
  editing.value = false
  e.target.value = model.value
}

const box = computed(() => (editing.value ? undefined : model.value))

const clamp = (n) => Math.min(props.max, Math.max(props.min, Number(n) || 0))

/* 数字框里显示什么由 text 说了算（绑 undefined 会把输入框清空，那就成了「打什么都看不见」）：
   · 拖拖动条 / 外部改值 → 回灌，但正在打字时不回灌，否则输到一半会被改掉
   · 自己在打字 → 原样显示，半截（空 / 只有一个减号）或超范围也先显示着
   · 失焦才归位 —— 输 9999 会显示成真正生效的 3840，打字期间不跳 */
const text = ref('')
const typing = ref(false)

watch(model, (v) => {
  if (!typing.value) text.value = String(v ?? '')
}, { immediate: true })

function onType(e) {
  const raw = e.target.value
  text.value = raw
  if (raw === '' || raw === '-') return
  const n = Number(raw)
  if (Number.isFinite(n)) model.value = clamp(n)
}

function onBlur() {
  typing.value = false
  text.value = String(model.value ?? '')
}

### 共享层（改这些要两个仓库一起改）

- **数字框的显示文本自己管**：`Slider` 的数字框用「本地 `text` + 打字期间不回灌 + 失焦归位」，**不要**把 `:value` 绑成 `undefined` —— 那会把输入框清空，成了「点进去打什么都看不见」（2026-11-10 修过）。拖动条与数字框靠同一个 `model` 双向同步。
- **选图的热区只有缩略图**：缩略图写 `<button type="button" @click="el?.click()">` + 一个 `sr-only` 的 `<input type="file">`，**别**把整行包进 `<label>` —— 那样点文字、点空白都会弹选图（视频封面 / 合集封面的背景行 2026-11-10 改过；角色一图流的缩略图本来就是这种写法）。
