let urlTimer = 0
function onUrlInput(slot, value) {
  setDraft(slot, value)
  clearTimeout(urlTimer)
  urlTimer = setTimeout(() => {
    if (draftOf(slot)) commitUrl(activeRow.value, slot)
  }, 700)
}

/* 粘完地址：先落代号 / 标记，再拿代号去源里反查，把缺的字段补齐 */
async function commitUrl(row, slot) {
  clearTimeout(urlTimer)
  const value = draftOf(slot)
  setDraft(slot, '')
  if (!value || !row) return

  const code = applyUrl(row, slot, value)
  if (!code) return
  /* 只给角色反查补齐：怪物同一个代号常被好几条共用，反查会带回别的条目的名字 */
  if (tab.value === 'characters' && needFill(row)) await hydrate(row, slot)
}

function onSelect(row) {
  urlDraft.value = { key: '', value: '' }
  select(row)
}

/* 改完名字：这条还缺什么就顺手从源补上 */
async function commitName(row) {

  if (tab.value === 'characters' && needFill(row)) await hydrate(row, 'name')
}

function onThumbError(e) {
  const img = e.target
  const alt = img.dataset.alt
  if (!alt) {
    img.removeAttribute('src')
    return
  }
  delete img.dataset.alt
  img.src = alt
}

async function onSync() {
  syncing.value = true
  try {
    const res = await syncFromAmber()
    if (!res) return
    if (res.conflicts.length) {
      conflicts.value = res.conflicts
      useSource.value = res.conflicts.map(() => false)
      conflictOpen.value = true
    } else {
      reportOpen.value = true
    }
  } finally {
    syncing.value = false
  }
}

function confirmConflicts() {
  const picked = conflicts.value.filter((_, i) => useSource.value[i])
  conflictOpen.value = false
  applyConflicts(picked)
  conflicts.value = []
}

   提交完把草稿清掉，框里回落到「已保存的值」，所以不会变成一个空框。 */
async function commitUrl(row, slot) {
  clearTimeout(urlTimer)
  const k = `${activeKey.value}|${slot}`
  const typing = draft.value.key === k
  const value = typing ? draft.value.value : ''
  draft.value = { key: '', value: '' }
  if (!row || !typing) return
  /* 清空输入框不算删数据（要删请用右边那个「清空」或框里的 ×），
     否则手一滑整格就没了 */
  if (!value.trim()) return

  const code = applyUrl(row, slot, value)
  reselect(row)
  if (!code) return
  /* 只给角色反查补齐：怪物同一个代号常被好几条共用，反查会带回别的条目的名字 */
  if (tab.value === 'characters' && needFill(row)) {
    await hydrate(row, slot)
    reselect(row)
  }
}

import { reactive, readonly } from 'vue'

/* 模块级单例：任何组件调 useToast() 拿到的是同一份队列 */

const items = reactive([])
let seq = 0

const KEEP = 3
const DEFAULT_DURATION = 3200
const ERROR_DURATION = 5000

export function useToast() {
  function remove(id) {
    const i = items.findIndex((t) => t.id === id)
    if (i < 0) return
    if (items[i].timer) clearTimeout(items[i].timer)
    items.splice(i, 1)
  }

  function push(text, opts = {}) {
    const type = opts.type ?? 'info'
    const duration = opts.duration ?? (type === 'error' ? ERROR_DURATION : DEFAULT_DURATION)
    const id = ++seq

    items.push({
      id,
      text,
      type,
      timer: duration > 0 ? setTimeout(() => remove(id), duration) : null,
    })

    /* 超出上限先淘汰最早的，避免连点堆成一屏 */
    while (items.length > KEEP) remove(items[0].id)

    return id
  }

  return {
    toasts: readonly(items),
    remove,
    show: push,
    success: (text, opts) => push(text, { ...opts, type: 'success' }),
    error: (text, opts) => push(text, { ...opts, type: 'error' }),
  }
}

import { ElMessage } from 'element-plus'

/* 全站提示：内部直接交给 EP 的 ElMessage（顶部消息、同样的文案会自动合并、不会连点堆一屏）。
   调用方仍然写 useToast().success(...) / .error(...)，一处都没改 ——
   这样 ToastHost 那个自造浮层可以整个删掉。 */

const DEFAULT_DURATION = 3200
const ERROR_DURATION = 5000

export function useToast() {
  function push(text, opts = {}) {
    const type = opts.type ?? 'info'
    const duration = opts.duration ?? (type === 'error' ? ERROR_DURATION : DEFAULT_DURATION)
    ElMessage({
      message: String(text ?? ''),
      type,
      duration,
      /* 同一句连续触发时合并，不叠一堆 */
      grouping: true,
      /* 让开顶栏（el-header 56px） */
      offset: 64,
    })
    return 0
  }

  return {
    show: push,
    success: (text, opts) => push(text, { ...opts, type: 'success' }),
    error: (text, opts) => push(text, { ...opts, type: 'error' }),
  }
}
